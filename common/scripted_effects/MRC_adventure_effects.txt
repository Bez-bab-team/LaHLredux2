MRC_TRPG_adventure_roll = {
 set_temp_variable = { dice_player = MRC_player_val }
 set_temp_variable = { dice_type = MRC_player_stat }
 meta_effect = {
  text = {
   set_temp_variable = { dice_mod = MRC_[GET_DICE_PLAYER]_stat^dice_type }
   set_temp_variable = { dice_prominent = 1 }
   if = {
    limit = { check_variable = { dice_mod < MRC_[GET_DICE_PLAYER]_stat^0 } }
    set_temp_variable = { dice_prominent = 0 }
   }
   if = {
    limit = { check_variable = { dice_mod < MRC_[GET_DICE_PLAYER]_stat^1 } }
    set_temp_variable = { dice_prominent = 0 }
   }
   if = {
    limit = { check_variable = { dice_mod < MRC_[GET_DICE_PLAYER]_stat^2 } }
    set_temp_variable = { dice_prominent = 0 }
   }
   if = {
    limit = { check_variable = { dice_mod < MRC_[GET_DICE_PLAYER]_stat^3 } }
    set_temp_variable = { dice_prominent = 0 }
   }
  }
  GET_DICE_PLAYER = "[MRC_TRPG_dice_player]"
 }
 if = {
  limit = {
   check_variable = { MRC_adventure_assurance = 1 }
   check_variable = { dice_prominent = 1 }
  }
  set_temp_variable = { dice_mod = 4 }
 }
 set_variable = { MRC_adventure_roll_bonus = dice_mod }
 set_temp_variable_to_random = { var = dice_raw min = 1 max = 21 integer = yes }
 set_variable = { MRC_adventure_roll_raw = dice_raw }
 add_to_temp_variable = { dice_raw = dice_mod }
 clamp_temp_variable = { var = dice_raw min = 1 max = 20 }
 set_variable = { MRC_adventure_roll_total = dice_raw }
 if = {
  limit = { check_variable = { dice_raw > 15 } }
  set_variable = { MRC_adventure_roll_result = 3 }
 }
 else_if = {
  limit = { check_variable = { dice_raw > 10 } }
  set_variable = { MRC_adventure_roll_result = 2 }
 }
 else_if = {
  limit = { check_variable = { dice_raw > 5 } }
  set_variable = { MRC_adventure_roll_result = 1 }
 }
 else = { set_variable = { MRC_adventure_roll_result = 0 } }
 set_variable = { MRC_adventure_assurance = 0 }
}

 if = {
  limit = {
   MRC_TRPG_can_complete_1A = yes
   has_country_flag = MRC_adventure_1A_started
   NOT = { has_country_flag = MRC_adventure_1A_main_pending }
  }

MRC_adventure_1A_roll = {
 if = {
  limit = { has_country_flag = MRC_adventure_1A_bonus_pending }
  clr_country_flag = MRC_adventure_1A_bonus_pending
  set_temp_variable = { MRC_player_val = 0 }
  set_temp_variable = { MRC_player_stat = 3 }
  MRC_TRPG_adventure_roll = yes
  set_variable = { MRC_adventure_1A_roll_result = MRC_adventure_roll_result }
  set_country_flag = MRC_adventure_1A_result_pending
  if = {
   limit = { check_variable = { MRC_adventure_roll_result = 3 } }
   country_event = { id = MRC_adventure.6 hours = 1 }
  }
  else_if = {
   limit = { check_variable = { MRC_adventure_roll_result = 2 } }
   country_event = { id = MRC_adventure.5 hours = 1 }
  }
  else_if = {
   limit = { check_variable = { MRC_adventure_roll_result = 1 } }
   country_event = { id = MRC_adventure.4 hours = 1 }
  }
  else = { country_event = { id = MRC_adventure.3 hours = 1 } }
 }
}
