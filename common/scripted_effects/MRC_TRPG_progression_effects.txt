MRC_TRPG_initialize_progression = {
	if = {
		limit = {
			original_tag = MRC
			has_country_flag = MRC_trpg_enabled
			NOT = { has_country_flag = MRC_trpg_progression_initialized }
		}
		clear_array = MRC_location_alignment
		resize_array = { MRC_location_alignment = 6 }
		set_variable = { MRC_party_pearce = 2 }
		set_variable = { MRC_party_moynihan = 2 }
		set_variable = { MRC_party_tibet = 1 }
		set_variable = { MRC_party_balance = 1 }
		set_variable = { MRC_party_boyd_rice = 0 }
		set_variable = { MRC_party_lord_miles = 0 }
		set_variable = { MRC_party_charles_bronson = 0 }
		set_variable = { MRC_party_vivianne_crowley = 0 }
		set_variable = { MRC_party_alan_moore = 0 }
		set_variable = { MRC_party_sheffield_officer = 0 }
		set_variable = { MRC_charles_bronson_role = 0 }
		set_country_flag = MRC_trpg_progression_initialized
	}
	MRC_TRPG_refresh_progress = yes
}

MRC_TRPG_refresh_progress = {
	if = {
		limit = { MRC_TRPG_progression_ready = yes }
		if = {
			limit = { check_variable = { MRC_point_control^0 = 0 } }
			set_variable = { MRC_point_control^0 = 1 }
		}
		if = {
			limit = {
				check_variable = { MRC_point_control^0 = 2 }
				check_variable = { MRC_point_control^1 = 0 }
			}
			set_variable = { MRC_point_control^1 = 1 }
		}
		else_if = {
			limit = {
				NOT = { check_variable = { MRC_point_control^0 = 2 } }
				check_variable = { MRC_point_control^1 = 1 }
			}
			set_variable = { MRC_point_control^1 = 0 }
		}
		if = {
			limit = { check_variable = { MRC_point_control^2 = 0 } }
			set_variable = { MRC_point_control^2 = 1 }
		}
		if = {
			limit = {
				check_variable = { MRC_point_control^2 = 2 }
				check_variable = { MRC_point_control^3 = 0 }
			}
			set_variable = { MRC_point_control^3 = 1 }
		}
		else_if = {
			limit = {
				NOT = { check_variable = { MRC_point_control^2 = 2 } }
				check_variable = { MRC_point_control^3 = 1 }
			}
			set_variable = { MRC_point_control^3 = 0 }
		}
		if = {
			limit = { check_variable = { MRC_point_control^4 = 0 } }
			set_variable = { MRC_point_control^4 = 1 }
		}
		if = {
			limit = {
				check_variable = { MRC_point_control^4 = 2 }
				check_variable = { MRC_point_control^5 = 0 }
			}
			set_variable = { MRC_point_control^5 = 1 }
		}
		else_if = {
			limit = {
				NOT = { check_variable = { MRC_point_control^4 = 2 } }
				check_variable = { MRC_point_control^5 = 1 }
			}
			set_variable = { MRC_point_control^5 = 0 }
		}
		set_variable = { MRC_locations_completed = 0 }
		set_variable = { MRC_left_choices = 0 }
		set_variable = { MRC_right_choices = 0 }
		if = {
			limit = { check_variable = { MRC_point_control^0 = 2 } }
			set_country_flag = MRC_TRPG_1A_completed
			add_to_variable = { MRC_locations_completed = 1 }
			if = {
				limit = { check_variable = { MRC_location_alignment^0 = 1 } }
				add_to_variable = { MRC_left_choices = 1 }
			}
			else_if = {
				limit = { check_variable = { MRC_location_alignment^0 = 2 } }
				add_to_variable = { MRC_right_choices = 1 }
			}
		}
		else = {
			clr_country_flag = MRC_TRPG_1A_completed
		}
		if = {
			limit = {
				check_variable = { MRC_point_selection = 0 }
				NOT = { check_variable = { MRC_point_control^0 = 1 } }
			}
			set_variable = { MRC_point_selection = 6 }
		}
		if = {
			limit = { check_variable = { MRC_point_control^1 = 2 } }
			set_country_flag = MRC_TRPG_1B_completed
			add_to_variable = { MRC_locations_completed = 1 }
			if = {
				limit = { check_variable = { MRC_location_alignment^1 = 1 } }
				add_to_variable = { MRC_left_choices = 1 }
			}
			else_if = {
				limit = { check_variable = { MRC_location_alignment^1 = 2 } }
				add_to_variable = { MRC_right_choices = 1 }
			}
		}
		else = {
			clr_country_flag = MRC_TRPG_1B_completed
		}
		if = {
			limit = {
				check_variable = { MRC_point_selection = 1 }
				NOT = { check_variable = { MRC_point_control^1 = 1 } }
			}
			set_variable = { MRC_point_selection = 6 }
		}
		if = {
			limit = { check_variable = { MRC_point_control^2 = 2 } }
			set_country_flag = MRC_TRPG_2A_completed
			add_to_variable = { MRC_locations_completed = 1 }
			if = {
				limit = { check_variable = { MRC_location_alignment^2 = 1 } }
				add_to_variable = { MRC_left_choices = 1 }
			}
			else_if = {
				limit = { check_variable = { MRC_location_alignment^2 = 2 } }
				add_to_variable = { MRC_right_choices = 1 }
			}
		}
		else = {
			clr_country_flag = MRC_TRPG_2A_completed
		}
		if = {
			limit = {
				check_variable = { MRC_point_selection = 2 }
				NOT = { check_variable = { MRC_point_control^2 = 1 } }
			}
			set_variable = { MRC_point_selection = 6 }
		}
		if = {
			limit = { check_variable = { MRC_point_control^3 = 2 } }
			set_country_flag = MRC_TRPG_2B_completed
			add_to_variable = { MRC_locations_completed = 1 }
			if = {
				limit = { check_variable = { MRC_location_alignment^3 = 1 } }
				add_to_variable = { MRC_left_choices = 1 }
			}
			else_if = {
				limit = { check_variable = { MRC_location_alignment^3 = 2 } }
				add_to_variable = { MRC_right_choices = 1 }
			}
		}
		else = {
			clr_country_flag = MRC_TRPG_2B_completed
		}
		if = {
			limit = {
				check_variable = { MRC_point_selection = 3 }
				NOT = { check_variable = { MRC_point_control^3 = 1 } }
			}
			set_variable = { MRC_point_selection = 6 }
		}
		if = {
			limit = { check_variable = { MRC_point_control^4 = 2 } }
			set_country_flag = MRC_TRPG_3A_completed
			add_to_variable = { MRC_locations_completed = 1 }
			if = {
				limit = { check_variable = { MRC_location_alignment^4 = 1 } }
				add_to_variable = { MRC_left_choices = 1 }
			}
			else_if = {
				limit = { check_variable = { MRC_location_alignment^4 = 2 } }
				add_to_variable = { MRC_right_choices = 1 }
			}
		}
		else = {
			clr_country_flag = MRC_TRPG_3A_completed
		}
		if = {
			limit = {
				check_variable = { MRC_point_selection = 4 }
				NOT = { check_variable = { MRC_point_control^4 = 1 } }
			}
			set_variable = { MRC_point_selection = 6 }
		}
		if = {
			limit = { check_variable = { MRC_point_control^5 = 2 } }
			set_country_flag = MRC_TRPG_3B_completed
			add_to_variable = { MRC_locations_completed = 1 }
			if = {
				limit = { check_variable = { MRC_location_alignment^5 = 1 } }
				add_to_variable = { MRC_left_choices = 1 }
			}
			else_if = {
				limit = { check_variable = { MRC_location_alignment^5 = 2 } }
				add_to_variable = { MRC_right_choices = 1 }
			}
		}
		else = {
			clr_country_flag = MRC_TRPG_3B_completed
		}
		if = {
			limit = {
				check_variable = { MRC_point_selection = 5 }
				NOT = { check_variable = { MRC_point_control^5 = 1 } }
			}
			set_variable = { MRC_point_selection = 6 }
		}
		if = {
			limit = { MRC_TRPG_all_locations_complete = yes }
			set_country_flag = MRC_TRPG_all_locations_completed
		}
		else = {
			clr_country_flag = MRC_TRPG_all_locations_completed
		}
		MRC_TRPG_refresh_party = yes
	}
}

MRC_TRPG_refresh_party = {
	if = {

		set_variable = { MRC_left_party_size = 0 }
		set_variable = { MRC_right_party_size = 0 }
		if = {
			limit = { check_variable = { MRC_party_pearce = 1 } }
			add_to_variable = { MRC_left_party_size = 1 }
		}

			limit = { check_variable = { MRC_party_pearce = 2 } }

		}
		if = {
			limit = { check_variable = { MRC_party_charles_bronson = 1 } }
			add_to_variable = { MRC_left_party_size = 1 }
		}
		else_if = {

		}
		set_variable = { MRC_party_boyd_rice = 2 }
		MRC_TRPG_refresh_party = yes
	}
